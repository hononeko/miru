1. Check that Miru is running:
   kubectl -n {{ .Release.Namespace }} rollout status deployment/{{ include "miru.fullname" . }}

2. Get the application URL:
{{- if .Values.httpRoute.enabled }}
{{- range .Values.httpRoute.hostnames }}
   https://{{ . }}
{{- end }}
{{- else if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
   http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else }}
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "miru.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
   Then open http://127.0.0.1:{{ .Values.service.port }}
{{- end }}

{{- if and (not .Values.existingSecret) (ne (lower .Values.auth.mode) "none") (lt (len .Values.auth.sessionSecret) 32) }}

WARNING: auth.sessionSecret is unset or shorter than 32 characters.
Miru requires AUTH_SESSION_SECRET (>= 32 characters) when auth.mode != "none".
Set --set auth.sessionSecret="$(openssl rand -base64 48)" or provide existingSecret.
{{- end }}

{{- if and (eq (lower .Values.database.engine) "sqlite") (gt (int .Values.replicaCount) 1) }}

WARNING: replicaCount is {{ .Values.replicaCount }} with database.engine=sqlite.
Miru stores live session state in memory and SQLite does not support multi-pod writes; set replicaCount=1 or switch to PostgreSQL.
{{- end }}
