1. Check that Miru is running:
   kubectl -n {{ .Release.Namespace }} rollout status deployment/{{ include "miru.fullname" . }}

2. Get the application URL:
{{- if .Values.httpRoute.enabled }}
{{- range .Values.httpRoute.hostnames }}
   https://{{ . }}
{{- end }}
{{- else if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
   http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if eq .Values.service.type "NodePort" }}
   export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "miru.fullname" . }})
   export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo "http://${NODE_IP}:${NODE_PORT}"
{{- else if eq .Values.service.type "LoadBalancer" }}
   NOTE: It may take a few minutes for the LoadBalancer IP to be available.
   You can watch the status by running:
     kubectl -n {{ .Release.Namespace }} get svc -w {{ include "miru.fullname" . }}
   export SERVICE_IP=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "miru.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
   echo "http://${SERVICE_IP}:{{ .Values.service.port }}"
{{- else }}
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "miru.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
   Then open http://127.0.0.1:{{ .Values.service.port }}
{{- end }}

{{- if and .Values.metrics.enabled (include "miru.dedicatedMetricsPort" .) }}

3. Prometheus metrics are served inside the cluster only, at:
   http://{{ include "miru.metricsServiceName" . }}.{{ .Release.Namespace }}.svc:{{ .Values.service.metricsPort }}/metrics
{{- end }}

{{- if and (include "miru.persistenceEnabled" .) .Values.persistence.keep (not .Values.persistence.existingClaim) }}

NOTE: `helm uninstall` keeps the {{ include "miru.pvcName" . }} PersistentVolumeClaim and the playback history on it.
Delete it with: kubectl -n {{ .Release.Namespace }} delete pvc {{ include "miru.pvcName" . }}
{{- end }}
